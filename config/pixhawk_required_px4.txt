# Parameter failsafe PX4 (Pixhawk 6C / Auterion) — dibaca app/pixhawk_check.py.
# Firmware dipilih otomatis dari HEARTBEAT (autopilot=12 -> file ini,
# autopilot=3 -> pixhawk_required.txt untuk ArduPilot).
#
# Format per baris (habis '#' = komentar):
#     NAMA_PARAM  MIN  MAX  KRITIS(y/n)
#     MIN/MAX '-' = tanpa batas. Nilai di luar [MIN..MAX] = pelanggaran.
#
# Cek HANYA MEMBACA (PARAM_REQUEST_READ) dan MELAPORKAN — tidak pernah menulis.
# Bila melanggar, ubah lewat QGC lalu ulangi.
#
# 'y' = kritis (jangan uji air bila melanggar); 'n' = advisory.

# Data-link (GCS/QGC/kabel telemetri) hilang -> aksi failsafe.
# 0=Disabled (BAHAYA, kapal jalan terus), 1=Hold, 2=Return, 3=Land.
# (4=Terminate / 5=Disarm mematikan motor — tidak untuk kapal berjalan.)
NAV_DLL_ACT      1  3  y
# RC remote hilang -> aksi failsafe. Enum sama dengan NAV_DLL_ACT.
NAV_RCL_ACT      1  3  y
# Baterai low -> aksi. 0=Warning saja (BAHAYA), 1=Return, 2=Land, 3=...
COM_LOW_BAT_ACT  1  3  n
# Ambang baterai low sebagai FRAKSI kapasitas (default 0.12 = 12%).
BAT_LOW_THR      0.05 0.30 n
# Ambang baterai critical sebagai FRAKSI (default 0.07); harus < BAT_LOW_THR.
BAT_CRIT_THR     0.03 0.20 n
# Aksi saat menembus geofence. 0=None (BAHAYA), 1=Warning, 2=Hold, 3=RTL.
GF_ACTION        1  3  n
